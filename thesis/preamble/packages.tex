% The class already loads fontspec, geometry, graphicx, booktabs, longtable,
% tabularx, caption, hyperref, amssymb, babel and fancyhdr, so only what it
% leaves out belongs here.

% Float placement, used by the figures that must stay inside their section.
\usepackage{float}

% Mathematics, used by the algorithm and metric definitions.
\usepackage{amsmath}

% Source listings for the appendix.
\usepackage{listings}
\usepackage{xcolor}

% Tables and figures are numbered within their chapter, as the department
% template allows.
\usepackage{chngcntr}
\counterwithin{table}{chapter}
\counterwithin{figure}{chapter}

% Bibliography. The department template prescribes IEEE style, numeric
% citations in square brackets, which is what the class's own sample document
% uses. Neither biblatex nor natbib is an option here: the class loads
% multibib, which biblatex refuses to load beside and which sends natbib into
% an infinite loop at \bibliography.
\bibliographystyle{IEEEtran}

% The class leaves hyperref boxes on by default. Links stay black and unboxed,
% as in the template's sample document.
\hypersetup{hidelinks,unicode}
