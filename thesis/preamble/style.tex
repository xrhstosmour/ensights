% Body text is justified at 1.5 line spacing, as is conventional for the
% department's submissions.
\onehalfspacing
\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt plus 2pt minus 1pt}

% Headings. The department template numbers chapters plainly, as "1.Title",
% without the report class's chapter-word prefix and without forcing each
% chapter onto a fresh page break style of its own.
\usepackage{titlesec}

\titleformat{\chapter}[hang]
  {\normalfont\Large\bfseries}{\thechapter.}{0.5em}{}
\titlespacing*{\chapter}{0pt}{0pt}{18pt}

\titleformat{\section}[hang]
  {\normalfont\large\bfseries}{\thesection}{0.5em}{}
\titlespacing*{\section}{0pt}{18pt}{9pt}

\titleformat{\subsection}[hang]
  {\normalfont\normalsize\bfseries}{\thesubsection}{0.5em}{}
\titlespacing*{\subsection}{0pt}{14pt}{7pt}

% Front matter headings that are not numbered chapters, for example the
% acknowledgements and the abstracts, share the chapter look but stay out of
% the numbering sequence while still appearing in the table of contents.
\newcommand{\unnumberedchapter}[1]{%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{#1}%
}

% Section names, matching the department template's exact wording.
%
% The timing here is fiddly. polyglossia \def's \captionsgreek to a variant
% macro (\monogreekcaptions and friends) while processing \setmainlanguage in
% the preamble, then *executes* it later when the language is actually
% selected, which happens after the begin-document hooks. So a plain
% \renewcommand, whether in the preamble or in \AtBeginDocument, is always
% overwritten, and appending in the preamble is discarded by that \def.
% Appending to \captionsgreek does not survive either, because polyglossia
% re-\def's it to the variant macro on every language selection. The variant
% macro itself, \monogreekcaptions for the default monotonic Greek, is defined
% once when gloss-greek.ldf loads and is never redefined, so appending there
% is what actually sticks. Revisit this if the Greek variant ever changes to
% polytonic or ancient, which use \polygreekcaptions/\ancientgreekcaptions.
\gappto\monogreekcaptions{%
  \renewcommand{\contentsname}{ΠΙΝΑΚΑΣ ΠΕΡΙΕΧΟΜΕΝΩΝ}%
  \renewcommand{\listtablename}{ΚΑΤΑΛΟΓΟΣ ΠΙΝΑΚΩΝ}%
  \renewcommand{\listfigurename}{ΚΑΤΑΛΟΓΟΣ ΔΙΑΓΡΑΜΜΑΤΩΝ/ΕΙΚΟΝΩΝ}%
  \renewcommand{\bibname}{ΒΙΒΛΙΟΓΡΑΦΙΑ}%
  \renewcommand{\appendixname}{ΠΑΡΑΡΤΗΜΑ}%
  \renewcommand{\tablename}{Πίνακας}%
  \renewcommand{\figurename}{Εικόνα}%
}

% Table captions sit below the table in the department template, so
% \caption is always placed after the tabular body in this document.

% Appendices are lettered with Greek capitals rather than the
% Latin A, B, C that \Alph produces. Note that the department template skips
% the digamma-style sixth letter and just continues through the alphabet.
\newcommand{\greekalphnum}[1]{%
  \ifcase\value{#1}\or Α\or Β\or Γ\or Δ\or Ε\or Ζ\or Η\or Θ\or Ι\or Κ\or
  Λ\or Μ\or Ν\or Ξ\or Ο\or Π\or Ρ\or Σ\or Τ\or Υ\or Φ\or Χ\or Ψ\or Ω\fi
}

% Source listings, used in the appendix.
\definecolor{listingbackground}{gray}{0.96}
\definecolor{listingcomment}{rgb}{0.25,0.45,0.25}
\definecolor{listingkeyword}{rgb}{0.20,0.20,0.65}
\definecolor{listingstring}{rgb}{0.60,0.25,0.15}

\lstdefinestyle{thesis}{
  backgroundcolor = \color{listingbackground},
  basicstyle      = \ttfamily\footnotesize,
  commentstyle    = \color{listingcomment},
  keywordstyle    = \color{listingkeyword}\bfseries,
  stringstyle     = \color{listingstring},
  breaklines      = true,
  captionpos      = b,
  frame           = single,
  showstringspaces = false,
  tabsize         = 4,
  % Listings is not Unicode-aware by default, so Greek inside a listing
  % needs these explicit mappings.
  extendedchars   = true,
  inputencoding   = utf8,
}
\lstset{style=thesis}
