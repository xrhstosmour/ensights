% The class owns fonts, geometry, spacing, headers and heading styles, and
% babel supplies the Greek caption names the department template uses
% (ΠΕΡΙΕΧΟΜΕΝΑ, ΚΑΤΑΛΟΓΟΣ ΣΧΗΜΑΤΩΝ, ΚΑΤΑΛΟΓΟΣ ΠΙΝΑΚΩΝ, Σχήμα, Πίνακας).
% Only what the class leaves undefined lives here.

% The manuscript marks inline English terms with \textenglish, which is a
% polyglossia command. The class uses babel, so the same name is kept as a
% thin wrapper and every call site stays untouched.
\newcommand{\textenglish}[1]{\foreignlanguage{english}{#1}}

% The manuscript cites with the biblatex command name. Plain bibtex offers
% one shape, the bracketed number, so the alias keeps the call sites unchanged.
\newcommand{\parencite}[1]{\cite{#1}}

% Unnumbered chapters that still appear in the table of contents, used by the
% glossary. The class provides the same thing for the abbreviations table.
\newcommand{\unnumberedchapter}[1]{%
  \chapter*{#1}%
  \addcontentsline{toc}{chapter}{#1}%
}

% Appendices are lettered with Greek capitals rather than the
% Latin A, B, C that \Alph produces. Note that the department template skips
% the digamma-style sixth letter and just continues through the alphabet.
\newcommand{\greekalphnum}[1]{%
  \ifcase\value{#1}\or Α\or Β\or Γ\or Δ\or Ε\or Ζ\or Η\or Θ\or Ι\or Κ\or
  Λ\or Μ\or Ν\or Ξ\or Ο\or Π\or Ρ\or Σ\or Τ\or Υ\or Φ\or Χ\or Ψ\or Ω\fi
}

% Source listings, used in the appendix.
\definecolor{listingbackground}{gray}{0.96}
\definecolor{listingcomment}{rgb}{0.25,0.45,0.25}
\definecolor{listingkeyword}{rgb}{0.20,0.20,0.65}
\definecolor{listingstring}{rgb}{0.60,0.25,0.15}

\lstdefinestyle{thesis}{
  backgroundcolor = \color{listingbackground},
  basicstyle      = \ttfamily\footnotesize,
  commentstyle    = \color{listingcomment},
  keywordstyle    = \color{listingkeyword}\bfseries,
  stringstyle     = \color{listingstring},
  breaklines      = true,
  captionpos      = b,
  frame           = single,
  showstringspaces = false,
  tabsize         = 4,
  % Listings is not Unicode-aware by default, so Greek inside a listing
  % needs these explicit mappings.
  extendedchars   = true,
  inputencoding   = utf8,
}
\lstset{style=thesis}

% listings keeps its English caption name under babel.
\renewcommand{\lstlistingname}{Κώδικας}
